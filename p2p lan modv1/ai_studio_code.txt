name: Build FreeServer Mod

on:
  push:
    branches: [ "**" ]
  pull_request:
    branches: [ "**" ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Depoyu Klonla
        uses: actions/checkout@v4

      - name: Java 21 Kur
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '21'

      - name: Gradle Ortamını Kur
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.10.2'

      - name: Modu Derle
        run: gradle build --no-daemon

      - name: Derlenen Jar Modunu Artifact Olarak Yükle
        uses: actions/upload-artifact@v4
        with:
          name: free_server-fabric-1.21.1
          path: build/libs/*.jar
          if-no-files-found: error
